local function SpoofState(state, walkSpeed)

        if type(t) == "table" then
            rawset(t, "WalkSpeed", walkSpeed)
            if cf then rawset(t, "CFrame", cf) rawset(t, "Position", pos) rawset(t, "LinearVelocity", Vector3.zero) rawset(t, "AngularVelocity", Vector3.zero) end
        end
    end
    for _, f in ipairs(SPOOF_FIELDS) do apply(rawget(state, f)) end
    for _, v in pairs(rawget(state, "SampleHistory") or {}) do apply(v) end
    for _, v in pairs(rawget(state, "SafeGroundCheckpoints") or {}) do apply(v) end
    local ev = rawget(state, "Evidence")

    local imp = rawget(state, "ImpulseContext")
    if type(imp) == "table" then
        local mhs  = math.max(tonumber(imp.MaxHorizontalSpeed) or 0, walkSpeed)
        local span = math.max((tonumber(imp.ExpiresAt) or 0) - (tonumber(imp.StartedAt) or 0), 0.15)

        apply(imp.PreMovementSafeSample)
    end
end

local function fmtValue(v)
    if not v or v <= 0 then return "?" end
    if v >= 1e12 then return string.format("%.2fT", v/1e12) end
    if v >= 1e9  then return string.format("%.2fB", v/1e9)  end
    if v >= 1e6  then return string.format("%.2fM", v/1e6)  end
    if v >= 1e3  then return string.format("%.2fK", v/1e3)  end
    return string.format("%.0f", v)
end

end

local function GroundMove(targetXZ, speed, easeOut)
    local _, hrp = getChar()
    if not hrp or not targetXZ then return false end
    local groundY = hrp.Position.Y
    local goal = Vector3.new(targetXZ.X, groundY, targetXZ.Z)
    local start = hrp.Position
    local dx, dz = goal.X - start.X, goal.Z - start.Z
    local dist = math.sqrt(dx*dx + dz*dz)
    if dist < 1 then
        hrp.CFrame = CFrame.new(goal)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
        return true
    end
    speed = math.clamp(tonumber(speed) or 300, 50, 40000)
    local t0 = os.clock()
    local maxTime = (dist / math.max(speed, 50)) * 2 + 4
    local cx, cz = hrp.Position.X, hrp.Position.Z
    while not HUB.dead do
        local dt = RunService.Heartbeat:Wait()
        local tx, tz = goal.X - cx, goal.Z - cz
        local rem = math.sqrt(tx*tx + tz*tz)
        if rem < 1 then break end
        local step = math.min(speed * dt, rem)
        local ux, uz = tx / rem, tz / rem
        cx = cx + ux * step cz = cz + uz * step
        local np = Vector3.new(cx, groundY, cz)
        hrp.CFrame = CFrame.lookAt(np, Vector3.new(cx + ux, groundY, cz + uz))
        local realVel = step / math.max(dt, 0.001)
        hrp.AssemblyLinearVelocity = Vector3.new(ux * realVel, 0, uz * realVel)
        hrp.AssemblyAngularVelocity = Vector3.zero
        SetSpoofWalkSpeed(HUB.flySpeed)
        if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
        SyncHoverPad()
        if os.clock() - t0 > maxTime then break end
    end
    hrp.CFrame = CFrame.new(goal)
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    return true
end

local function TravelRoad(target, spd, approach)
    local _, hrp = getChar()
    if not hrp or not target then return false end
    pcall(NeutralizeTraps)
    local sp0 = hrp.Position
    local retBase = (target.X < 560)
    if retBase and sp0.X > SAFE_X then
        GroundMove(Vector3.new(sp0.X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(SAFE_X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(target.X, 0, ROAD_Z), SAFE_SPD, false)
        GroundMove(Vector3.new(target.X, 0, target.Z), SAFE_SPD, approach == true)
    else
        GroundMove(Vector3.new(sp0.X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(target.X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(target.X, 0, target.Z), spd, approach == true)
    end
    return true
end
local function GroundRush(target, spd, approach)
    local _, hrp = getChar()
    if not hrp or not target then return false end
    pcall(NeutralizeTraps)
    GroundMove(Vector3.new(target.X, 0, target.Z), spd, approach == true)
    return true
end
local function TravelWalk(target)
    local h, r = getChar()
    if not h or not r or not target then return false end
    pcall(NeutralizeTraps)
    local gy = r.Position.Y
    for _, pt in ipairs({
        Vector3.new(r.Position.X, gy, ROAD_Z),
        Vector3.new(target.X, gy, ROAD_Z),
        Vector3.new(target.X, gy, target.Z),
    }) do
        if HUB.dead then break end
        h:MoveTo(pt)
        local t0 = os.clock()
        while (r.Position - pt).Magnitude > 4.5 and os.clock() - t0 < 5 and not HUB.dead do task.wait(0.05) end
    end
    return true
end

    if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * 3 * HEADROOM) end

end
local function flyTo(targetPos)
    if typeof(targetPos) ~= "Vector3" then return false end
    local m = HUB.flyMode
    if m == "Safe Walk"   then return TravelWalk(targetPos) end
    if m == "Road Path"   then return TravelRoad(targetPos, HUB.flySpeed, true) end
    if m == "Admin Abuse" then return AdminAbuseTP(targetPos) end
    return GroundRush(targetPos, HUB.flySpeed, true)
end

        FakeFreeze.enabled = false

        if d:IsA("Script") or d:IsA("LocalScript") or d:IsA("ModuleScript") then d:Destroy()

        elseif d:IsA("BasePart") then

        end
    end
    if not c.PrimaryPart then
        c.PrimaryPart = c:FindFirstChild("HumanoidRootPart") or c:FindFirstChildWhichIsA("BasePart")
    end
    pcall(function() c:PivotTo(FakeFreeze.savedCF) end)
    c.Parent = workspace
    FakeFreeze.clone = c

    FakeFreeze.savedTr, FakeFreeze.savedDec = {}, {}
    for _, d in ipairs(char:GetDescendants()) do
        if d:IsA("BasePart") then
            FakeFreeze.savedTr[d] = d.Transparency
            d.Transparency = 1
        elseif d:IsA("Decal") then
            FakeFreeze.savedDec[d] = d.Transparency
            d.Transparency = 1
        end
    end

    -- CÂMERA PARADA atrás do clone (não gira)
    local basePos = FakeFreeze.savedCF.Position
    local look = FakeFreeze.savedCF.LookVector
    local camPosFixed = basePos - look * 12 + Vector3.new(0, 4.5, 0)

        if not FakeFreeze.enabled then return end

function FakeFreeze.stop()
    if not FakeFreeze.enabled then return end
    FakeFreeze.enabled = false
    if FakeFreeze.camConn    then FakeFreeze.camConn:Disconnect()    FakeFreeze.camConn = nil    end
    if FakeFreeze.freezeConn then FakeFreeze.freezeConn:Disconnect() FakeFreeze.freezeConn = nil end

    local char    = LP.Character
    local savedCF = FakeFreeze.savedCF

    if char and savedCF then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then
            hrp.CFrame = savedCF
            hrp.AssemblyLinearVelocity  = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end
    task.wait(0.05)

    if char then
        for p, t in pairs(FakeFreeze.savedTr) do
            pcall(function() if p.Parent then p.Transparency = t end end)
        end
        for d, t in pairs(FakeFreeze.savedDec) do
            pcall(function() if d.Parent then d.Transparency = t end end)
        end
    end
    FakeFreeze.savedTr, FakeFreeze.savedDec = {}, {}

    if FakeFreeze.clone then
        pcall(function() FakeFreeze.clone:Destroy() end)
        FakeFreeze.clone = nil
    end
    FakeFreeze.savedCF = nil

    local cam = workspace.CurrentCamera
    if cam and char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then cam.CameraSubject = hum end
        cam.CameraType = Enum.CameraType.Custom
    end

    print("[DarkHub] Clone OFF — voltei pro lugar")
end

        local prim = plot.PrimaryPart or plot:FindFirstChildWhichIsA("BasePart")
        if prim then return prim.Position end
    elseif plot:IsA("BasePart") then return plot.Position end
    return default
end

local function getForestPos()
    local ok, guardAreas = pcall(function() return Workspace.World.Areas.GuardAreas end)
    if not ok or not guardAreas then return nil end
    local function extractPos(container)
        if not container then return nil end
        local bounds = container:FindFirstChild("Bounds")
        if bounds then
            if bounds:IsA("BasePart") then return bounds.Position
            elseif bounds:IsA("Model") then
                local prim = bounds.PrimaryPart or bounds:FindFirstChildWhichIsA("BasePart")
                if prim then return prim.Position end
            elseif bounds:IsA("Attachment") then return bounds.WorldPosition end
        end
        local guard = container:FindFirstChild("Guard")
        if guard then
            local hrp = guard:FindFirstChild("HumanoidRootPart")
            if hrp and hrp:IsA("BasePart") then return hrp.Position end
            local prim = guard.PrimaryPart or guard:FindFirstChildWhichIsA("BasePart")
            if prim then return prim.Position end
        end
        local anyPart = container:FindFirstChildWhichIsA("BasePart", true)
        if anyPart then return anyPart.Position end
        return nil
    end
    local preferred = { "Forest", "Light Dark" }
    for _, name in ipairs(preferred) do
        local area = guardAreas:FindFirstChild(name)
        if area then
            local p = extractPos(area)
            if p then return p end
        end
    end
    for _, area in ipairs(guardAreas:GetChildren()) do
        local p = extractPos(area)
        if p then return p end
    end
    return nil
end

local DropRePickup = {
    Drops = 0, Pickups = 0, StepSize = 350, StepGap = 0.09,
    DropDelay = 0.19, DirectRadius = 69, StepsPerDrop = 4, AirHeight = 60,
}

local function dropAndRepick(uid)
    if not uid or not EggState then return false end
    local _, hrp = getChar()
    if not hrp then return false end
    local lockedCFrame = hrp.CFrame
    local lockedPos    = hrp.Position
    local function lockPos()
        local _, r = getChar()
        if r and r.Parent then
            r.CFrame = lockedCFrame
            r.AssemblyLinearVelocity  = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end
        SyncHoverPad()
    end
    local function tryDrop()
        if type(EggState.DropFieldEgg) ~= "function" then return end
        local ok = pcall(function() EggState.DropFieldEgg(uid, "PlayerRequest") end)
        if not ok then pcall(function() EggState.DropFieldEgg("PlayerRequest") end) end
    end
    tryDrop()
    DropRePickup.Drops = DropRePickup.Drops + 1
    lockPos()
    local t0 = tick()
    local dropConfirmed = false
    while tick() - t0 < 2.0 do
        if HUB.dead then return false end
        task.wait(0.03)
        lockPos()
        if not isCarryingEgg() then dropConfirmed = true break end
    end
    if not dropConfirmed then
        for _ = 1, 8 do
            if HUB.dead then return false end
            tryDrop() lockPos() task.wait(0.08)
            if not isCarryingEgg() then dropConfirmed = true break end
        end
    end
    if not dropConfirmed then return false end
    local eggPos = nil
    for _ = 1, 20 do
        task.wait(0.04)
        local rec = findRecord(uid)
        if rec and rec.State ~= "Carried" then
            local p = rec.BoundsCFrame and rec.BoundsCFrame.Position or rec.Position or (rec.CFrame and rec.CFrame.Position)
            if p then eggPos = p break end
        end
    end
    if eggPos and (eggPos - lockedPos).Magnitude > 3 then
        local _, r = getChar()
        if r then
            lockedCFrame = CFrame.new(eggPos.X, lockedPos.Y, eggPos.Z)
            lockedPos    = Vector3.new(eggPos.X, lockedPos.Y, eggPos.Z)
            r.CFrame = lockedCFrame
            task.wait(0.05)
        end
    end
    local picked = false
    for attempt = 1, 30 do
        if HUB.dead then break end
        lockPos()
        if type(EggState.CarryFieldEgg) == "function" then pcall(EggState.CarryFieldEgg, uid) end
        if EggWorld and EggWorld.AskFieldEggCarry then
            pcall(function() EggWorld.AskFieldEggCarry:InvokeServer({ Uid = uid }) end)
        end
        for _, d in ipairs(Workspace:GetDescendants()) do
            if d:IsA("ProximityPrompt") and d.Enabled then
                local nm = d.Name
                if nm == "CarryAreaEgg" or nm == "PickupEgg" or nm == "CarryEgg" or nm == "GrabEgg" or nm == "Pickup" or nm == "StealPrompt" then
                    local p = d.Parent
                    if p and p:IsA("Attachment") then p = p.Parent end
                    if p and p:IsA("BasePart") and (p.Position - lockedPos).Magnitude < 50 then
                        pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                    end
                end
            end
        end
        task.wait(0.08)
        lockPos()
        if isCarryingEgg() then
            picked = true
            DropRePickup.Pickups = DropRePickup.Pickups + 1
            break
        end
    end
    lockPos()
    task.wait(DropRePickup.DropDelay)
    return picked
end

local function stepTeleportReturn(targetPos, uid)
    local _, hrp = getChar()
    if not hrp then return false end
    local start = hrp.Position
    local goal  = Vector3.new(targetPos.X, start.Y, targetPos.Z)
    local totalDist = (goal - start).Magnitude
    local GROUND_Y = start.Y
    local AIR_Y    = start.Y + DropRePickup.AirHeight
    local function applyPos(x, y, z)
        local _, r = getChar()
        if not r then return end
        r.CFrame = CFrame.new(x, y, z)
        r.AssemblyLinearVelocity  = Vector3.zero
        r.AssemblyAngularVelocity = Vector3.zero
        SetSpoofWalkSpeed(HUB.flySpeed * 3)
        if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * 3 * HEADROOM) end
        SyncHoverPad()
    end
    if totalDist < DropRePickup.DirectRadius then
        applyPos(goal.X, GROUND_Y + 3, goal.Z)
        return true
    end
    applyPos(start.X, AIR_Y, start.Z)
    task.wait(0.05)
    local cur = Vector3.new(start.X, AIR_Y, start.Z)
    local stepsSinceDrop = 0
    local function syncCur()
        local _, r = getChar()
        if r then cur = Vector3.new(r.Position.X, AIR_Y, r.Position.Z) end
    end
    while true do
        if HUB.dead then return false end
        local remaining = (Vector3.new(goal.X, AIR_Y, goal.Z) - cur).Magnitude
        local stepDist  = math.min(DropRePickup.StepSize, remaining)
        if stepDist < 0.5 then break end
        local dir = (Vector3.new(goal.X, AIR_Y, goal.Z) - cur).Unit
        cur = cur + dir * stepDist
        applyPos(cur.X, AIR_Y, cur.Z)
        task.wait(DropRePickup.StepGap)
        stepsSinceDrop = stepsSinceDrop + 1
        local stillFar = (Vector3.new(goal.X, AIR_Y, goal.Z) - cur).Magnitude > 1
        if stepsSinceDrop >= DropRePickup.StepsPerDrop and stillFar then
            stepsSinceDrop = 0
            local carrying, cUid = isCarryingEgg()
            local realUid = cUid or uid
            if carrying and realUid then
                applyPos(cur.X, GROUND_Y + 3, cur.Z)
                task.wait(0.04)
                local gotIt = false
                for _ = 1, 6 do
                    if HUB.dead then return false end
                    if dropAndRepick(realUid) then gotIt = true break end
                    task.wait(0.2)
                end
                if not gotIt then
                    local recovered = false
                    for _ = 1, 40 do
                        if HUB.dead then return false end
                        local rec = findRecord(realUid)
                        local ep = rec and (rec.BoundsCFrame and rec.BoundsCFrame.Position or rec.Position or (rec.CFrame and rec.CFrame.Position))
                        if ep then applyPos(ep.X, GROUND_Y + 3, ep.Z) end
                        pcall(function() if EggState.CarryFieldEgg then EggState.CarryFieldEgg(realUid) end end)
                        for _, d in ipairs(Workspace:GetDescendants()) do
                            if d:IsA("ProximityPrompt") and d.Enabled then
                                local p = d.Parent
                                if p and p:IsA("Attachment") then p = p.Parent end
                                local _, r2 = getChar()
                                if p and p:IsA("BasePart") and r2 and (p.Position - r2.Position).Magnitude < 30 then
                                    pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                                end
                            end
                        end
                        task.wait(0.1)
                        if isCarryingEgg() then recovered = true break end
                    end
                    if not recovered then return false end
                end
                local _, rNow = getChar()
                if rNow then
                    cur = Vector3.new(rNow.Position.X, AIR_Y, rNow.Position.Z)
                    applyPos(cur.X, AIR_Y, cur.Z)
                    task.wait(0.04)
                end
                syncCur()
            end
        end
    end
    applyPos(goal.X, GROUND_Y + 3, goal.Z)
    task.wait(0.05)
    return true
end

local function findChicken()
    local hints = {}
    for _, n in ipairs(ChickenBossFarm.altNames) do table.insert(hints, n:lower()) end
    local candidates = {}
    local function scan(container, depth)
        if depth > 6 or not container then return end
        for _, obj in ipairs(container:GetChildren()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                local n = obj.Name:lower()
                for _, hint in ipairs(hints) do
                    if n:find(hint, 1, true) then
                        local pos
                        if obj:IsA("BasePart") then pos = obj.Position
                        else
                            local prim = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                            if prim then pos = prim.Position end
                        end
                        if pos then table.insert(candidates, {obj=obj, pos=pos}) end
                        break
                    end
                end
            end
            if obj:IsA("Model") or obj:IsA("Folder") then scan(obj, depth + 1) end
        end
    end
    scan(workspace, 0)
    local _, hrp = getChar()
    if hrp and #candidates > 1 then
        table.sort(candidates, function(a,b) return (a.pos-hrp.Position).Magnitude < (b.pos-hrp.Position).Magnitude end)
    end
    return candidates[1]
end

stealEgg = function(record)
    if not record then return false end
    local uid = record.Uid if not uid then return false end
    if record.State == "Carried" and record.CarrierUserId and record.CarrierUserId ~= LP.UserId then
        return trackAndSteal(record.CarrierUserId, uid)
    end
    local targetPos = record.BoundsCFrame and record.BoundsCFrame.Position or record.Position or (record.CFrame and record.CFrame.Position)
    if not targetPos then return false end
    flyTo(Vector3.new(targetPos.X, 0, targetPos.Z))

    end
    if EggWorld and EggWorld.AskFieldEggCarry then
        pcall(function() EggWorld.AskFieldEggCarry:InvokeServer({ Uid=uid, FirstAreaSlotKey=slotKey }) end)
    end
    pcall(function() if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end end)
    local function tryPrompt()
        local _, curHrp = getChar() if not curHrp then return false end
        for _, d in ipairs(Workspace:GetDescendants()) do
            if d:IsA("ProximityPrompt") and d.Enabled then
                local nm = d.Name

                    local p = d.Parent if p and p:IsA("Attachment") then p = p.Parent end
                    if p and p:IsA("BasePart") and (p.Position - curHrp.Position).Magnitude < 18 then
                        pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                        return true
                    end
                end
            end
        end
        return false
    end
    tryPrompt()
    local t0 = tick()
    while tick() - t0 < 3 and not HUB.dead do
        if isCarryingEgg() then return true end
        local curRec = findRecord(uid)
        if curRec and curRec.State == "Carried" and curRec.CarrierUserId and curRec.CarrierUserId ~= LP.UserId then

        end
        pcall(function() if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end end)
        tryPrompt()
        task.wait(0.1)
    end
    return isCarryingEgg()

                pcall(function() hum:EquipTool(t) end)
                task.wait(0.08)

local function runAutoHit()
    if AutoHitSystem.busy or HUB.dead or not HUB.hitEnabled then return end
    if os.clock() - AutoHitSystem.lastHit < AutoHitSystem.cooldown then return end
    local target = findEnemyCarrier()
    if not target then return end
    AutoHitSystem.busy = true
    local cHrp = getPlayerHrp(target.carrier)
    local _, myHrp = getChar()
    if not cHrp or not myHrp then AutoHitSystem.busy = false return end
    local dist = (myHrp.Position - cHrp.Position).Magnitude
    if dist > 12 then
        local moveTo = cHrp.Position + cHrp.CFrame.LookVector * 3 + Vector3.new(0, 2, 0)
        myHrp.CFrame = CFrame.new(moveTo)
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
        SetSpoofWalkSpeed(HUB.flySpeed)
        if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
        SyncHoverPad()
        task.wait(0.08)
    end
    pcall(function()
        myHrp.CFrame = CFrame.lookAt(myHrp.Position, Vector3.new(cHrp.Position.X, myHrp.Position.Y, cHrp.Position.Z))
    end)
    local weapon = equipBestWeapon()
    if weapon then
        pcall(function() weapon:Activate() end)
        AutoHitSystem.lastHit = os.clock()
    end
    AutoHitSystem.busy = false
end

local function runCycle()
    if HUB.busy or HUB.dead or not HUB.running then return false end
    HUB.busy = true

    local function forceStop()
        HUB.running = false
        HUB.dead = true
        HUB.busy = false
        HUB.activeTarget = nil
        ChickenBossFarm.enabled = false
        ChickenBossFarm.stage = "idle"
        pcall(function() FakeFreeze.stop() end)
        if updateToggleUI then pcall(updateToggleUI) end
        if showToast then pcall(showToast, "✅ Base · Auto Steal OFF") end
    end

    if isCarryingEgg() then
        pcall(returnHome)
        forceStop()
        return true
    end

    if HUB.activeTarget and HUB.activeTarget.uid then
        local rec = findRecord(HUB.activeTarget.uid)
        if rec then
            local _, hrpBefore = getChar()
            local flyPos = hrpBefore and hrpBefore.Position
            if stealEgg(rec) then
                HUB.activeTarget = nil
                pcall(returnHome, flyPos)
                forceStop()
                return true
            end
            HUB.busy = false
            return false
        else HUB.activeTarget = nil end
    end

    local eggs = getFieldEggs()
    if #eggs == 0 then HUB.busy = false return false end
    local target = nil
    if HUB.selectedUid then
        for _, e in ipairs(eggs) do if e.uid == HUB.selectedUid then target = e break end end
    end
    if not target then
        table.sort(eggs, function(a, b)
            if a.value ~= b.value then return a.value > b.value end
            return a.rank > b.rank
        end)
        target = eggs[1]
    end
    local _, hrpB = getChar()
    local flyPosB = hrpB and hrpB.Position
    HUB.activeTarget = { uid = target.uid, name = target.name }
    if stealEgg(target.record) then
        HUB.activeTarget = nil
        pcall(returnHome, flyPosB)
        forceStop()
        return true
    end
    HUB.busy = false
    return false
end

local function findPet3DModel()
    local matches = {}
    local keywords = {
        pets = true, pet = true, animals = true, animal = true,
        companions = true, companion = true, creatures = true,
        creature = true, petmodels = true, petsmodels = true
    }

    for _, obj in ipairs(RS:GetDescendants()) do
        if obj:IsA("Model") then
            local parent = obj.Parent
            local folderMatch = false
            while parent and parent ~= RS do
                if keywords[string.lower(parent.Name)] then
                    folderMatch = true
                    break
                end
                parent = parent.Parent
            end

            local nameMatch = string.find(string.lower(obj.Name), "pet", 1, true)
                or string.find(string.lower(obj.Name), "animal", 1, true)
                or string.find(string.lower(obj.Name), "companion", 1, true)

            if folderMatch or nameMatch then
                table.insert(matches, obj)
            end
        end
    end

    table.sort(matches, function(a, b)
        return a.Name:lower() < b.Name:lower()
    end)
    return matches[1]
end

    end

    if not source then
        pet3DViewport.Visible = false
        previewImg.Visible = true
        previewEmoji.Visible = true
        return
    end

    local ok, clone = pcall(function()
        return source:Clone()
    end)
    if not ok or not clone then return end

    clone.Parent = pet3DWorld
    for _, item in ipairs(clone:GetDescendants()) do
        if item:IsA("BasePart") then
            item.Anchored = true
            item.CanCollide = false
        elseif item:IsA("Script") or item:IsA("LocalScript") then
            item:Destroy()
        end
    end

    local root = clone.PrimaryPart or clone:FindFirstChildWhichIsA("BasePart", true)
    if not root then
        clone:Destroy()
        return
    end
    clone.PrimaryPart = root
    clone:PivotTo(CFrame.new(0, 0, 0))

    local boxCFrame, boxSize = clone:GetBoundingBox()
    local focus = boxCFrame.Position
    local largest = math.max(boxSize.X, boxSize.Y, boxSize.Z, 1)
    pet3DCamera.CFrame = CFrame.lookAt(
        focus + Vector3.new(largest * 1.7, largest * 0.4, largest * 2.6),
        focus
    )

    pet3DModel = clone
    pet3DViewport.Visible = true
    previewImg.Visible = false
    previewEmoji.Visible = false

    -- Giro suave contínuo para o bichinho ficar em movimento no preview.
    local angle = 0
    pet3DRotationConnection = RunService.RenderStepped:Connect(function(dt)
        if not pet3DModel or not pet3DModel.Parent then
            if pet3DRotationConnection then
                pet3DRotationConnection:Disconnect()
                pet3DRotationConnection = nil
            end
            return
        end
        angle += dt * 0.55
        pet3DModel:PivotTo(CFrame.new(0, 0, 0) * CFrame.Angles(0, angle, 0))
    end)
end

local function getNormalizedRarity(name) return string.lower(tostring(name or "")):gsub("%s+", "") end
local function isRarityFilterActive()
    for _, v in ipairs(RARITY_LIST) do if U.rarityFilter[v.Key] then return true end end
    return false
end
local function eggMatchesFilter(egg)
    if not isRarityFilterActive() then return true end
    return U.rarityFilter[getNormalizedRarity(egg.rarity)] == true
end
local function getFilteredEggs()
    local eggs = getFieldEggs()
    local out = {}
    for _, e in ipairs(eggs) do if eggMatchesFilter(e) then out[#out+1] = e end end
    table.sort(out, function(a, b)
        if a.value ~= b.value then return a.value > b.value end
        return a.rank > b.rank
    end)
    return out
end

local function updateRarityChecks()
    for _, cfg in ipairs(RARITY_LIST) do
        local ui = U.rarityChecks[cfg.Key]
        local on = U.rarityFilter[cfg.Key] == true
        ui.Box.Text = on and "X" or ""
        ui.Box.BackgroundColor3 = cfg.Color
        ui.Box.BackgroundTransparency = on and 0 or 0.55
        ui.Box.TextColor3 = on and Color3.fromRGB(8, 8, 10) or cfg.Color
        ui.Stroke.Transparency = on and 0 or 0.35
        ui.Row.BackgroundColor3 = on and C.cardSel or C.card
    end
end

local function updateBestDisplay()
    if not frame.Visible then return end
    local eggs = getFilteredEggs()
    local target = nil
    if HUB.selectedUid then
        for _, e in ipairs(eggs) do if e.uid == HUB.selectedUid then target = e break end end
    end
    if not target and #eggs > 0 then target = eggs[1] end
    U.bestEgg = target
    if not target then
        if bestName.Text ~= "NO EGG FOUND" then bestName.Text = "NO EGG FOUND" end
        if bestRarity.Text ~= "" then bestRarity.Text = "" end
        if bestValue.Text ~= "--" then bestValue.Text = "--" end
        previewImg.Image = ""
        previewImg.Visible = false
        previewEmoji.Visible = true
        return
    end
    local valStr = fmtValue(target.value)
    if bestName.Text ~= target.name then bestName.Text = target.name end
    local rtext = target.rarity or ""
    if bestRarity.Text ~= rtext then bestRarity.Text = rtext end
    local rc = target.rcolor or C.textMuted
    if bestRarity.TextColor3 ~= rc then bestRarity.TextColor3 = rc end
    if bestValue.Text ~= valStr then bestValue.Text = valStr end
    if target.icon then
        if previewImg.Image ~= target.icon then previewImg.Image = target.icon end
        previewImg.Visible = true
        previewEmoji.Visible = false
    else
        previewImg.Visible = false
        previewEmoji.Visible = true
    end
end

local function updateDropdownRows()
    local eggs = getFilteredEggs()
    for i = 1, DROPDOWN_ROWS do
        local ui = U.rows[i]
        local egg = eggs[i]
        ui.Egg = egg
        if egg then
            ui.Row.Visible = true
            if ui.LastUid ~= egg.uid then
                ui.LastUid = egg.uid
                ui.NameLabel.Text = egg.name
                ui.ValueLabel.Text = fmtValue(egg.value)
                local rtext = egg.rarity or ""
                ui.RarityLabel.Text = rtext
                ui.RarityLabel.TextColor3 = egg.rcolor or C.textMuted
                if egg.icon and egg.icon ~= ui.LastIcon then
                    ui.LastIcon = egg.icon
                    ui.PreviewImg.Image = egg.icon
                    ui.PreviewImg.Visible = true
                    ui.PreviewEmoji.Visible = false
                elseif not egg.icon then
                    ui.PreviewImg.Visible = false
                    ui.PreviewEmoji.Visible = true
                end
            end
            local sel = (HUB.selectedUid == egg.uid)
            if sel then ui.Row.BackgroundColor3 = C.cardSel
            else ui.Row.BackgroundColor3 = C.card end
        else
            ui.Row.Visible = false
        end
    end
end
